_PIMGS={}   # dest file name -> JPEG bytes, written to dist/assets/people/ by make.py
def _photo_src(rel):
    rel=(rel or '').strip()
    if not rel: return None
    full=os.path.normpath(os.path.join('..',rel)); root=os.path.normpath('../data/media')
    if not full.startswith(root+os.sep) or not os.path.isfile(full):
        print('WARNING: photo not found or not in data/media, using initials instead:',rel); return None
    return full
def _square(src,dest):
    try:
        im=ImageOps.exif_transpose(Image.open(src)).convert('RGB')
    except Exception as ex:
        print('WARNING: could not read photo',src,ex); return False
    w,h=im.size; s=min(w,h)
    left=(w-s)//2; top=int((h-s)*0.2) if h>w else (h-s)//2
    im=im.crop((left,top,left+s,top+s)).resize((256,256),Image.LANCZOS)
    buf=io.BytesIO(); im.save(buf,'JPEG',quality=85,optimize=True); _PIMGS[dest]=buf.getvalue(); return True
def _wide(src,dest):
    try:
        im=ImageOps.exif_transpose(Image.open(src)).convert('RGB')
    except Exception as ex:
        print('WARNING: could not read photo',src,ex); return None
    if im.size[0]>1500: im=im.resize((1500,round(im.size[1]*1500/im.size[0])),Image.LANCZOS)
    buf=io.BytesIO(); im.save(buf,'JPEG',quality=82,optimize=True); _PIMGS[dest]=buf.getvalue(); return im.size
